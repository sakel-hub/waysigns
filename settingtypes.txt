# Display mode for the sign HUD:
# - waypoint: 3D in-world floating waypoint anchored directly at the sign.
# - overlay: Fixed 2D HUD element placed on the player's screen.
waysigns_display_mode (Display Mode) enum waypoint waypoint,overlay

# Maximum distance in blocks (meters) from the sign to show the HUD.
waysigns_max_distance (Max Detection Distance) float 4.5 1.0 15.0

# Polling interval in seconds for raycast checks (lower = more responsive, higher = lower CPU usage in multiplayer).
waysigns_check_interval (Check Interval) float 0.05 0.01 0.5

# Fade-in and fade-out animation duration in seconds (0.0 for instant display without fade transitions).
waysigns_fade_time (Fade Duration) float 0.3 0.0 1.0

# HUD overlay and typography scaling multiplier (e.g. 2.0 for 2x larger HUD overlay).
# Automatically adapts responsively on smaller screens and mobile resolutions.
waysigns_hud_scale (HUD Size Scale) float 2.0 1.0 3.5

# Match the HUD overlay width and height to the physical aspect ratio of the pointed sign.
# Ensures the background plaque and text layout authentically replicate the sign in the world.
waysigns_match_aspect_ratio (Match Sign Aspect Ratio) bool true

# Maximum characters per line before word wrapping.
waysigns_max_chars_per_line (Max Characters Per Line) int 30 15 60

# Maximum lines displayed simultaneously before auto-paginating long texts.
waysigns_max_lines (Max Visible Lines) int 5 2 10

# Automatically cycle through pages when text exceeds the max line limit.
waysigns_auto_scroll (Auto Scroll Long Texts) bool true

# Delay in seconds before advancing to the next page of long text.
waysigns_scroll_delay (Scroll Delay) float 5.0 1.0 15.0

# Show decorative plaque border and corner rivets around the dynamic texture background.
waysigns_show_frame (Show Decorative Frame) bool true

# Completely disable and remove attached text entities from 3rd party sign mods (signs_lib, basic_signs, etc.).
# Frees server memory, eliminates entity limits and lag spikes, and keeps sign nodes clean.
waysigns_disable_sign_entities (Disable 3rd Party Sign Entities) bool true

# Allow attached text entities specifically for street_signs mod even when entity suppression is enabled.
# Allows players to see 3D in-world text on street blades and highway signs while still using WaySigns.
waysigns_enable_street_signs_entities (Enable street_signs Entities) bool false

# Vertical screen position (Y coordinate 0.0 - 1.0) when using 2D overlay mode.
# Default 0.50 places the overlay directly in the center of the screen / crosshair.
waysigns_overlay_pos_y (Overlay Screen Y Position) float 0.50 0.2 0.8

# Text contrast mode:
# - auto: Automatically optimizes text luminance for dark or light sign backgrounds.
# - bright_text: Always forces bright high-visibility text on all backgrounds.
# - preserve: Uses the exact original text color without contrast adaptation.
waysigns_contrast_mode (Contrast Mode) enum auto auto,bright_text,preserve

# Background plaque darkening glaze intensity (0.0 = completely transparent natural texture, 0.8 = heavy dark glaze).
# Default 0.40 provides rich contrast and WCAG AAA legibility while keeping wood grain and textures clearly visible.
waysigns_background_darkness (Background Glaze Darkness) float 0.40 0.0 0.8

# Enable WaySigns HUD waypoints for all nodes with infotext (chests, furnaces, machines).
waysigns_enable_node_infotext (Enable Node Infotext HUD) bool true

# Infotext HUD overlay and typography scaling multiplier for nodes with infotext.
# Default 2.0 provides clear visibility and proportional quickview sizing.
waysigns_infotext_scale (Node Infotext Size Scale) float 2.0 0.8 3.5

# Vertical position offset in blocks for node infotext waypoints relative to node center.
# Default 0.35 places the infotext in the top half of the node (node spans -0.5 to +0.5).
waysigns_infotext_pos_y_offset (Node Infotext 3D Vertical Offset) float 0.35 0.0 0.5

# Vertical screen position (Y coordinate 0.0 - 1.0) when using 2D overlay mode for node infotext.
# Default 0.38 places the infotext in the upper half of the screen, above the crosshair.
waysigns_infotext_overlay_pos_y (Node Infotext Overlay Screen Y Position) float 0.38 0.1 0.9

# Enable small quickview of items stored in pointed container nodes (chests, furnaces, shelves).
waysigns_enable_inventory_quickview (Enable Container Inventory Quickview) bool true

# Maximum number of occupied item slots displayed in the background quickview dock/grid (1 to 32).
waysigns_quickview_max_slots (Quickview Maximum Item Slots) int 32 1 32

# Display all occupied items in the container up to maximum slots (in a clean multi-row grid, no empty slots).
waysigns_quickview_show_all (Quickview Show All Occupied Items) bool true

# Respect container locks and protection (suppress private inventory contents for non-owners).
waysigns_quickview_respect_locks (Quickview Respect Container Locks) bool true

# Enable the WaySigns Inscription Marker tool (waysigns:marker) for writing on nodes and entities.
waysigns_enable_marker (Enable Inscription Marker Tool) bool true

# Number of inscription uses before the marker tool wears out (0 = infinite durability).
waysigns_marker_uses (Marker Uses Before Depletion) int 100 0 1000

# Enable WaySigns HUD inspection and floating waypoints for inscribed entities.
waysigns_enable_entity_inspection (Enable Entity Inscription Inspection) bool true

# Maximum characters allowed per inscription when writing with the marker tool.
waysigns_marker_max_chars (Marker Inscription Max Characters) int 250 10 1000
